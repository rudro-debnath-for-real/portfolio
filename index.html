using namespace std;

class Solution {
public:
    int maxProfit(vector& prices) {
        int minPrice = INT_MAX;
        int maxProfit = 0;

        for (int price : prices) {
            minPrice = min(minPrice, price);
            maxProfit = max(maxProfit, price - minPrice);
        }
        return maxProfit;
    }
};

// Time: O(n) | Space: O(1)
// Greedy approach - track minimum
